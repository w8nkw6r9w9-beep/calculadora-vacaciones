<script>

  // Feriados Nacionales Chile (Agrega o actualiza fechasYYYY-MM-DD según corresponda)

  const feriadosChile = [

    // 2024

    "2024-01-01", "2024-03-29", "2024-05-01", "2024-05-21", "2024-06-20", 

    "2024-07-16", "2024-08-15", "2024-09-18", "2024-09-19", "2024-09-20", 

    "2024-10-31", "2024-11-01", "2024-12-25",

    // 2025

    "2025-01-01", "2025-04-18", "2025-05-01", "2025-05-21", "2025-06-20", 

    "2025-07-16", "2025-08-15", "2025-09-18", "2025-09-19", "2025-10-31", 

    "2025-12-08", "2025-12-25",

    // 2026

    "2026-01-01", "2026-04-03", "2026-05-01", "2026-05-21", "2026-06-29", 

    "2026-07-16", "2026-09-18", "2026-10-12", "2026-12-08", "2026-12-25"

  ];


  function esFeriado(fechaStr) {

    return feriadosChile.includes(fechaStr);

  }


  function calcularFeriado() {

    const inicioStr = document.getElementById('fInicio').value;

    const terminoStr = document.getElementById('fTermino').value;

    const sueldo = parseFloat(document.getElementById('sueldo').value);


    if (!inicioStr || !terminoStr || isNaN(sueldo) || sueldo <= 0) {

      alert("Por favor completa todos los campos correctamente.");

      return;

    }


    const inicio = new Date(inicioStr + 'T00:00:00');

    const termino = new Date(terminoStr + 'T00:00:00');


    if (termino < inicio) {

      alert("La fecha de término no puede ser anterior a la fecha de inicio.");

      return;

    }


    // 1. Calcular días totales trabajados

    const diffTime = Math.abs(termino - inicio);

    const diasTrabajados = Math.ceil(diffTime / (1000 * 60 * 60 * 24)) + 1; // Incluye ambos días


    // 2. Días hábiles acumulados (15 / 365 = 0.04109589 por día)

    const diasHabilesAcc = diasTrabajados * (15 / 365);


    // 3. Proyección en el calendario partiendo desde el día siguiente al término

    let fechaAux = new Date(termino);

    fechaAux.setDate(fechaAux.getDate() + 1);


    let habilesContados = 0;

    

    // Sumar días hábiles en el calendario saltando fines de semana y feriados

    while (habilesContados < diasHabilesAcc) {

      const diaSemana = fechaAux.getDay(); // 0 = Dom, 6 = Sáb

      const fechaIso = fechaAux.toISOString().split('T')[0];


      // Si es de lunes a viernes (1 a 5) y NO es feriado registrado

      if (diaSemana !== 0 && diaSemana !== 6 && !esFeriado(fechaIso)) {

        // Si falta menos de 1 día entero de fracción habiles

        if (diasHabilesAcc - habilesContados < 1) {

          habilesContados = diasHabilesAcc; // completado

          break;

        } else {

          habilesContados += 1;

        }

      }

      // Avanzar al siguiente día

      fechaAux.setDate(fechaAux.getDate() + 1);

    }


    // 4. Determinar días corridos a pagar (Diferencia de días entre fecha de término y fecha proyección)

    const diffCorridos = Math.abs(fechaAux - termino);

    let diasCorridos = Math.ceil(diffCorridos / (1000 * 60 * 60 * 24));


    // Si la fracción restante es menor a un día hábil completo, mantenemos precisión

    const fraccionDiasCorridos = (diasCorridos - 1) + (diasHabilesAcc % 1);


    // 5. Cálculo monetario

    const valorDiario = sueldo / 30;

    const totalPagar = fraccionDiasCorridos * valorDiario;


    // Mostrar resultados

    document.getElementById('diasHabiles').innerText = diasHabilesAcc.toFixed(2);

    document.getElementById('diasCorridos').innerText = fraccionDiasCorridos.toFixed(2);

    document.getElementById('montoDiario').innerText = '$' + Math.round(valorDiario).toLocaleString('es-CL');

    document.getElementById('montoTotal').innerText = '$' + Math.round(totalPagar).toLocaleString('es-CL');

    document.getElementById('resultados').style.display = 'block';

  }

</script>
